cmake_minimum_required(VERSION 3.22)
project(inventory_binding_v1_host LANGUAGES C CXX)
get_filename_component(DH2_BINDING_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_subdirectory("${DH2_BINDING_ROOT}/port/game-data" game-data)
get_target_property(selected dh2_game_data SOURCES)
set(found FALSE)
foreach(existing IN LISTS selected)
 get_filename_component(name "${existing}" NAME)
 if(name STREQUAL "player_equipment_queries_live_v1.cpp")
  set(found TRUE)
 endif()
endforeach()
if(NOT found)
 target_sources(dh2_game_data PRIVATE "${DH2_BINDING_ROOT}/port/game-data/player_equipment_queries_live_v1.cpp")
endif()
include("${DH2_BINDING_ROOT}/port/engine-ui/inventory_text_v1.cmake")
add_executable(inventory_binding_text_queries_v1 ../inventory_binding_text_queries_v1.cpp
 "${DH2_BINDING_ROOT}/port/pydata-constants/constants.c" "${DH2_BINDING_ROOT}/port/random/random.c")
target_link_libraries(inventory_binding_text_queries_v1 PRIVATE dh2_inventory_text_v1)
add_executable(inventory_binding_format_gold_v1 ../hud_text_format_v1.cpp)
target_link_libraries(inventory_binding_format_gold_v1 PRIVATE dh2_inventory_text_v1)
add_executable(inventory_binding_varargs_gold_v1 ../item_text_varargs_v5.cpp ../item_text_varargs_v5_fixture.cpp)
target_link_libraries(inventory_binding_varargs_gold_v1 PRIVATE dh2_inventory_text_v1)
add_executable(localization_level_up_title_v1 ../localization_level_up_title_v1.cpp
 "${DH2_BINDING_ROOT}/port/pydata-constants/constants.c")
target_link_libraries(localization_level_up_title_v1 PRIVATE dh2_inventory_text_v1)
