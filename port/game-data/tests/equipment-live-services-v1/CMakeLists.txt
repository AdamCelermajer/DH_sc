cmake_minimum_required(VERSION 3.22)
project(equipment_live_services_v1_host LANGUAGES C CXX)
get_filename_component(DH2_EQUIPMENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_subdirectory("${DH2_EQUIPMENT_ROOT}/port/game-data" game-data)
get_target_property(selected dh2_game_data SOURCES)
set(found FALSE)
foreach(existing IN LISTS selected)
 get_filename_component(name "${existing}" NAME)
 if(name STREQUAL "player_equipment_live_services_v1.cpp")
  set(found TRUE)
 endif()
endforeach()
if(NOT found)
 target_sources(dh2_game_data PRIVATE "${DH2_EQUIPMENT_ROOT}/port/game-data/player_equipment_live_services_v1.cpp")
endif()
set_source_files_properties("${DH2_EQUIPMENT_ROOT}/port/game-data/player_equipment_live_services_v1.cpp"
 TARGET_DIRECTORY dh2_game_data PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
add_executable(player_equipment_live_services_v1_selected ../player_equipment_live_services_v1.cpp
 "${DH2_EQUIPMENT_ROOT}/port/random/random.c")
target_link_libraries(player_equipment_live_services_v1_selected PRIVATE dh2_game_data)
target_compile_options(player_equipment_live_services_v1_selected PRIVATE -fno-fast-math -ffp-contract=off)
