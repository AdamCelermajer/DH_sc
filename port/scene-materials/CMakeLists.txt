cmake_minimum_required(VERSION 3.22)
project(dh2_scene_materials LANGUAGES CXX)
add_library(dh2_scene_materials SHARED scene.cpp ../engine-resources/resources.cpp ../asset-payloads/payloads.cpp ../engine-math/math.cpp)
target_compile_features(dh2_scene_materials PUBLIC cxx_std_17)
target_include_directories(dh2_scene_materials PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(dh2_scene_materials PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
if(ANDROID)
  target_link_options(dh2_scene_materials PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
else()
  enable_testing()
  add_executable(scene_audit tests/audit.cpp)
  target_link_libraries(scene_audit PRIVATE dh2_scene_materials)
endif()
