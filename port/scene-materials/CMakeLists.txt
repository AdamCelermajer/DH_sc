cmake_minimum_required(VERSION 3.22)
project(dh2_scene_materials LANGUAGES CXX)
add_library(dh2_scene_materials SHARED scene.cpp ../engine-resources/resources.cpp ../asset-payloads/payloads.cpp ../engine-math/math.cpp)
target_sources(dh2_scene_materials PRIVATE ../asset-payloads/sha256.cpp)
target_sources(dh2_scene_materials PRIVATE shader_sources.cpp)
target_sources(dh2_scene_materials PRIVATE blood_render_pass_v3.cpp transparent_queue_v1.cpp particle_scene_v1.cpp)
target_sources(dh2_scene_materials PRIVATE effect_render_pass_v4.cpp)
target_sources(dh2_scene_materials PRIVATE transparent_sort_v2.cpp)
target_sources(dh2_scene_materials PRIVATE shader_reflection_v4.cpp shader_program_collection_v4.cpp material_compare_v4.cpp effect_material_directory_v4.cpp)
target_sources(dh2_scene_materials PRIVATE swf_texture.cpp)
target_compile_features(dh2_scene_materials PUBLIC cxx_std_17)
target_include_directories(dh2_scene_materials PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(dh2_scene_materials PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
if(ANDROID)
  target_link_options(dh2_scene_materials PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
else()
  enable_testing()
  add_executable(scene_audit tests/audit.cpp)
  target_link_libraries(scene_audit PRIVATE dh2_scene_materials)
  add_executable(shader_sources_audit tests/shader_sources.cpp)
  target_link_libraries(shader_sources_audit PRIVATE dh2_scene_materials)
  add_executable(swf_texture_audit tests/swf_texture.cpp)
  target_link_libraries(swf_texture_audit PRIVATE dh2_scene_materials)
endif()
