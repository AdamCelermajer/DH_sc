cmake_minimum_required(VERSION 3.22)
project(dh2_engine_particles LANGUAGES CXX)
if(NOT TARGET dh2_scene_materials)
  add_subdirectory(../scene-materials "${CMAKE_CURRENT_BINARY_DIR}/scene-materials")
endif()
add_library(dh2_engine_particles SHARED particle_runtime.cpp)
target_compile_features(dh2_engine_particles PUBLIC cxx_std_17)
target_include_directories(dh2_engine_particles PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(dh2_engine_particles PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
target_link_libraries(dh2_engine_particles PUBLIC dh2_scene_materials)
if(ANDROID)
  target_link_options(dh2_engine_particles PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
else()
  add_executable(particle_runtime_fixture tests/particle_runtime_fixture.cpp)
  target_link_libraries(particle_runtime_fixture PRIVATE dh2_engine_particles)
endif()
