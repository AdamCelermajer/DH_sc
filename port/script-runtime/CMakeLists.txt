cmake_minimum_required(VERSION 3.22)
project(dh2_script_runtime LANGUAGES C CXX)
option(DH2_SCRIPT_SANITIZERS "Build host audit with address/undefined sanitizers" OFF)
set(LUA_CORE lapi lcode ldebug ldo ldump lfunc lgc llex lmem lobject lopcodes
  lparser lstate lstring ltable ltm lundump lvm lzio lauxlib lbaselib ltablib lstrlib lmathlib)
set(LUA_SOURCES)
foreach(unit IN LISTS LUA_CORE)
  list(APPEND LUA_SOURCES lua/${unit}.c)
endforeach()
add_library(dh2_script_runtime SHARED script_runtime.c script_game_bindings.c ${LUA_SOURCES})
target_sources(dh2_script_runtime PRIVATE script_function_alias.cpp)
target_compile_features(dh2_script_runtime PRIVATE cxx_std_17)
target_include_directories(dh2_script_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE lua)
target_compile_options(dh2_script_runtime PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(dh2_script_runtime PRIVATE m)
if(ANDROID)
  target_link_options(dh2_script_runtime PRIVATE -Wl,-z,max-page-size=16384)
else()
  add_executable(lua514_audit tests/lua514.cpp)
  target_link_libraries(lua514_audit PRIVATE dh2_script_runtime)
  target_compile_features(lua514_audit PRIVATE cxx_std_17)
  target_include_directories(lua514_audit PRIVATE lua)
  add_executable(script_game_bindings_audit tests/script_game_bindings.cpp ../level-world/character_timers.cpp)
  target_include_directories(script_game_bindings_audit PRIVATE ../level-world)
  target_link_libraries(script_game_bindings_audit PRIVATE dh2_script_runtime)
  target_compile_features(script_game_bindings_audit PRIVATE cxx_std_17)
  add_executable(script_game_bindings_replay tests/script_game_bindings_replay.cpp)
  target_link_libraries(script_game_bindings_replay PRIVATE dh2_script_runtime)
  target_compile_features(script_game_bindings_replay PRIVATE cxx_std_17)
  if(DH2_SCRIPT_SANITIZERS)
    foreach(target dh2_script_runtime lua514_audit script_game_bindings_audit script_game_bindings_replay)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
endif()
