cmake_minimum_required(VERSION 3.22)
project(dh2_nativeinterface_reconstruction C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(DH2_BUILD_JNI "Build JNI library (requires Android or desktop JDK headers)" ON)
option(DH2_BUILD_TESTS "Build host semantic tests" OFF)
add_library(nativeinterface SHARED nativeinterface.c)
target_compile_options(nativeinterface PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
if(DH2_BUILD_JNI)
  target_sources(nativeinterface PRIVATE nativeinterface_jni.c)
  if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(nativeinterface PRIVATE ${JNI_INCLUDE_DIRS})
  endif()
  target_link_options(nativeinterface PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
if(ANDROID)
  target_link_options(nativeinterface PRIVATE -Wl,-z,max-page-size=16384)
endif()
if(DH2_BUILD_TESTS)
  enable_testing()
  add_executable(nativeinterface_test tests/semantic_test.c nativeinterface.c)
  target_include_directories(nativeinterface_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(nativeinterface_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME nativeinterface_semantics COMMAND nativeinterface_test)
endif()
