cmake_minimum_required(VERSION 3.22)
project(dh2_stage_loader_v38 LANGUAGES CXX)
get_filename_component(loader "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(dh2_stage_v38_manager ../stage_loader_v38_manager_probe.cpp ../stage-loader-v38-manager-overlay/canonical_object_manager_v1.cpp)
target_include_directories(dh2_stage_v38_manager PRIVATE ../stage-loader-v38-manager-overlay "${loader}/vendor/character-rng-integration-v5-loading/port/level-world")
add_executable(dh2_stage_v38_trace ../stage_loader_v38_trace_probe.cpp)
target_include_directories(dh2_stage_v38_trace PRIVATE "${loader}")
add_executable(dh2_stage_v38_root ../stage_loader_v38_root_probe.cpp)
target_include_directories(dh2_stage_v38_root PRIVATE "${loader}")
add_executable(dh2_stage_v38_post ../stage-loader-v38-lifecycle-overlay/lifecycle_v36_test.cpp ../stage-loader-v38-lifecycle-overlay/lifecycle_v36.cpp)
target_include_directories(dh2_stage_v38_post PRIVATE ../stage-loader-v38-lifecycle-overlay)
add_executable(dh2_stage_v38_bridge ../stage_loader_v38_bridge_probe.cpp ../stage-loader-v38-manager-overlay/canonical_object_manager_v1.cpp)
target_include_directories(dh2_stage_v38_bridge PRIVATE ../stage-loader-v38-manager-overlay "${loader}/vendor/character-rng-integration-v5-loading/port/level-world" "${loader}")
foreach(target dh2_stage_v38_manager dh2_stage_v38_trace dh2_stage_v38_root dh2_stage_v38_post dh2_stage_v38_bridge)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(${target} PRIVATE -fsanitize=address,undefined)
endforeach()
