cmake_minimum_required(VERSION 3.22)
project(dh2_engine_textures LANGUAGES CXX)
add_library(dh2_engine_textures SHARED textures.cpp pvrtc.cpp)
target_compile_features(dh2_engine_textures PUBLIC cxx_std_17)
target_include_directories(dh2_engine_textures PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(dh2_engine_textures PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
if(ANDROID)
  target_link_options(dh2_engine_textures PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
endif()
if(NOT ANDROID)
  enable_testing()
  add_executable(texture_safety tests/safety.cpp)
  target_link_libraries(texture_safety PRIVATE dh2_engine_textures)
  add_test(NAME texture_safety COMMAND texture_safety)
endif()
